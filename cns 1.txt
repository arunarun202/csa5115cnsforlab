#include<stdio.h>
int main() {
    char s[50];
    int k, i;
    printf("Enter text: ");
    scanf("%s", s);
    printf("Enter key: ");
    scanf("%d", &k);
    for(i = 0; s[i] != '\0'; i++)
        s[i] = ((s[i] - 'a' + k) % 26) + 'a';
    printf("Encrypted text: %s", s);
    return 0;
}
