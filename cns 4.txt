#include<stdio.h>
#include<string.h>
int main()
{
    char t[100],k[20];
    int i,l;
    printf("Text: ");
    scanf("%s",t);
    printf("Key: ");
    scanf("%s",k);
    l=strlen(k);
    for(i=0;t[i];i++)
        t[i]=((t[i]-'A')+(k[i%l]-'A'))%26+'A';
    printf("Cipher: %s",t);
    return 0;
}
